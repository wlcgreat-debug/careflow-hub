### 피드백 데이터 구조

```python
class Feedback(BaseModel):
    id: str
    agent_id: str
    task_type: str  # "칼럼", "썸네일", "코드" 등
    
    original_output: str  # 에이전트가 처음 낸 것
    modified_output: str  # 세영님이 수정한 것
    
    changes: list[Change]  # 변경 사항 목록
    
    learned_rule: str  # 추출된 규칙
    # 예: "'-께서는' 대신 '-은/는' 사용"
    
    confidence: float  # 이 규칙의 확신도
    apply_count: int   # 이 규칙이 적용된 횟수
    success_rate: float  # 적용 후 승인율
    
    created_at: datetime
    last_applied: datetime

class Change(BaseModel):
    change_type: str  # "replace", "delete", "add", "reorder"
    before: str
    after: str
    context: str  # 주변 맥락
    category: str  # "톤", "용어", "구조", "길이" 등
```

```python
async def generate_with_learning(agent, task):
    # 1. 해당 에이전트의 학습된 규칙 조회
    rules = await get_learned_rules(agent.id, task.type)
    
    # 2. 규칙을 프롬프트에 주입
    learning_context = format_rules_for_prompt(rules)
    
    # 3. 생성
    output = await agent.generate(
        task=task,
        additional_context=f"""
        ## 이전 피드백에서 학습한 것들
        {learning_context}
        
        위 규칙들을 반영해서 작성해주세요.
        적용한 규칙이 있으면 말씀해주세요.
        """
    )
    
    # 4. 적용된 규칙 표시
    return output + "\n\n📚 적용한 학습 규칙: rule_001, rule_002"
```

### 메모리 데이터 구조

```python
class Memory(BaseModel):
    id: str
    type: Literal["project", "decision", "feedback", "conversation", "knowledge"]
    
    title: str
    summary: str  # 핵심 내용 요약
    full_content: str  # 전체 내용
    
    entities: list[str]  # 관련 엔티티 ["CareFlow 2.0", "React", "환자 대시보드"]
    tags: list[str]  # 검색용 태그
    
    importance: float  # 0-1, 중요도
    access_count: int  # 조회 횟수 (자주 보는 건 더 중요)
    
    created_at: datetime
    last_accessed: datetime
    expires_at: Optional[datetime]  # L2는 만료일 있음
    
    related_memories: list[str]  # 연관 메모리 ID
    
class Project(BaseModel):
    """프로젝트 단위 메모리"""
    id: str
    name: str
    status: Literal["planning", "active", "paused", "completed", "archived"]
    
    description: str
    goals: list[str]
    
    decisions: list[Decision]  # 주요 결정들
    participants: list[str]  # 참여 에이전트
    
    timeline: list[Milestone]
    
    created_at: datetime
    updated_at: datetime
    
class Decision(BaseModel):
    """결정 기록"""
    id: str
    project_id: str
    
    question: str  # "프론트엔드 프레임워크 뭘로?"
    decision: str  # "React로 결정"
    reasoning: str  # "기존 경험 + 생태계 고려"
    alternatives: list[str]  # ["Vue", "Svelte"]
    decided_by: str  # "세영님" or "아키텍트"
    
    created_at: datetime
```

### 메모리 검색 (RAG)

```python
async def recall_relevant_memories(query: str, context: dict) -> list[Memory]:
    """관련 메모리 검색"""
    
    # 1. 임베딩 기반 유사도 검색
    query_embedding = await embed(query)
    similar_memories = await vector_search(query_embedding, top_k=10)
    
    # 2. 엔티티 매칭
    entities = extract_entities(query)  # ["CareFlow 2.0"]
    entity_memories = await search_by_entities(entities)
    
    # 3. 현재 프로젝트 컨텍스트
    if context.get("current_project"):
        project_memories = await get_project_memories(context["current_project"])
    
    # 4. 병합 및 랭킹
    all_memories = merge_and_rank(similar_memories, entity_memories, project_memories)
    
    # 5. 상위 N개 반환
    return all_memories[:5]
```

```python
class Priority(BaseModel):
    score: float  # 0-100
    level: Literal["critical", "high", "medium", "low"]
    reasoning: str

def calculate_priority(task: Task) -> Priority:
    score = 0
    reasons = []
    
    # 1. 마감일 (40점)
    if task.deadline:
        hours_until = (task.deadline - now()).total_seconds() / 3600
        if hours_until < 1:
            score += 40
            reasons.append("1시간 내 마감")
        elif hours_until < 24:
            score += 30
            reasons.append("오늘 마감")
        elif hours_until < 72:
            score += 20
            reasons.append("3일 내 마감")
        else:
            score += 10
    
    # 2. 영향 범위 (25점)
    if task.affects_external:  # 외부 발행
        score += 25
        reasons.append("외부 영향")
    elif task.affects_users:  # 환자/구독자 영향
        score += 20
        reasons.append("사용자 영향")
    elif task.affects_team:  # 다른 에이전트 블로킹
        score += 15
        reasons.append("팀 블로킹")
    
    # 3. 세영님 직접 요청 (20점)
    if task.requested_by == "user":
        score += 20
        reasons.append("세영님 직접 요청")
    
    # 4. 의존성 (15점)
    blocking_count = count_blocked_tasks(task)
    if blocking_count > 2:
        score += 15
        reasons.append(f"{blocking_count}개 작업 블로킹")
    elif blocking_count > 0:
        score += 10
    
    # 레벨 결정
    if score >= 80:
        level = "critical"
    elif score >= 60:
        level = "high"
    elif score >= 40:
        level = "medium"
    else:
        level = "low"
    
    return Priority(
        score=score,
        level=level,
        reasoning=", ".join(reasons)
    )
```

```python
async def process_task_queue(queue: list[Task]):
    """우선순위 기반 작업 처리"""
    
    # 1. 우선순위로 정렬
    sorted_tasks = sorted(queue, key=lambda t: t.priority.score, reverse=True)
    
    # 2. Critical은 즉시 처리
    critical_tasks = [t for t in sorted_tasks if t.priority.level == "critical"]
    if critical_tasks:
        await process_immediately(critical_tasks[0])
        return
    
    # 3. 나머지는 에이전트별로 병렬 처리
    agent_tasks = group_by_agent(sorted_tasks)
    
    await asyncio.gather(*[
        process_agent_tasks(agent_id, tasks)
        for agent_id, tasks in agent_tasks.items()
    ])
```
